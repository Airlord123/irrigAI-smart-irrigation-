#ifndef SENSOR_MANAGER_H
#define SENSOR_MANAGER_H

#include <Arduino.h>
#include <DHT.h>
#include <Preferences.h>
#include "config.h"

class SensorManager {
public:
    SensorManager();

    // Initialize sensors and load calibration data from NVS
    void begin();

    // Read current physical parameters
    void update();

    // Getters for sensor metrics
    float getTemperature() const { return temperature; }
    float getHumidity() const { return humidity; }
    int getRawMoisture() const { return rawMoisture; }
    float getMoisturePercent() const { return moisturePercent; }

    // Calibration setters (writes to NVS and immediately updates runtime variables)
    void calibrateDry(int rawVal);
    void calibrateWet(int rawVal);

    // Get current calibration thresholds
    int getDryCalibrationValue() const { return dryValue; }
    int getWetCalibrationValue() const { return wetValue; }

private:
    DHT dht;
    Preferences preferences;

    // Runtime state variables
    float temperature;
    float humidity;
    int rawMoisture;
    float moisturePercent;

    // Calibration thresholds (loaded from NVS or default config)
    int dryValue;
    int wetValue;

    // Calculate moisture percentage based on calibration and raw values
    float calculateMoisturePercent(int raw) const;
};

#endif // SENSOR_MANAGER_H
