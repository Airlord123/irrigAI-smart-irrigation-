#include "pumpcontroller.h"

PumpController::PumpController() 
    : running(false),
      safetyCutoffActive(false),
      pumpStartTime(0),
      lastPumpOffTime(0) {}

void PumpController::begin() {
    pinMode(PUMP_PIN, OUTPUT);
    digitalWrite(PUMP_PIN, LOW); // Keep off initially
    
    running = false;
    safetyCutoffActive = false;
    pumpStartTime = 0;
    // Set last pump off time to a past point to allow immediate starting on boot
    lastPumpOffTime = 0; 
    
    Serial.println("[PumpController] Initialized.");
}

void PumpController::update() {
    if (running) {
        unsigned long elapsed = millis() - pumpStartTime;
        if (elapsed >= PUMP_MAX_RUN_TIME) {
            Serial.print("[PumpController] SAFETY: Max run time reached (");
            Serial.print(elapsed);
            Serial.println(" ms). Forcing Pump OFF!");
            
            turnOff();
            safetyCutoffActive = true; // Lock pump from auto restarting
        }
    }
}

void PumpController::turnOn(bool isManualOverride) {
    if (running) {
        return; // Already running
    }

    if (safetyCutoffActive && !isManualOverride) {
        Serial.println("[PumpController] Blocked: Safety cutoff is active. Clear error to run.");
        return;
    }

    if (!isManualOverride && !canRunAuto()) {
        Serial.println("[PumpController] Blocked: Pump is in cooldown cycle.");
        return;
    }

    // Reset safety cutoff if user manually overrides
    if (isManualOverride && safetyCutoffActive) {
        resetSafetyCutoff();
    }

    digitalWrite(PUMP_PIN, HIGH);
    running = true;
    pumpStartTime = millis();
    Serial.print("[PumpController] Pump turned ON. Manual Override: ");
    Serial.println(isManualOverride ? "YES" : "NO");
}

void PumpController::turnOff() {
    if (!running) {
        return; // Already off
    }

    digitalWrite(PUMP_PIN, LOW);
    running = false;
    lastPumpOffTime = millis();
    
    Serial.print("[PumpController] Pump turned OFF. Ran for: ");
    Serial.print(lastPumpOffTime - pumpStartTime);
    Serial.println(" ms");
}

bool PumpController::canRunAuto() const {
    if (safetyCutoffActive) {
        return false;
    }
    if (running) {
        return false;
    }
    // If we have watered before, respect the cooldown period
    if (lastPumpOffTime > 0 && (millis() - lastPumpOffTime < PUMP_COOLDOWN_TIME)) {
        return false;
    }
    return true;
}

void PumpController::resetSafetyCutoff() {
    if (safetyCutoffActive) {
        safetyCutoffActive = false;
        Serial.println("[PumpController] Safety cutoff cleared.");
    }
}

unsigned long PumpController::getRunningDuration() const {
    if (!running) {
        return 0;
    }
    return millis() - pumpStartTime;
}
