#include "edgeai.h"

// Struct representation for saving weights in a clean binary block in NVS
struct ModelWeights {
    float weightsIH[INPUT_NODES][HIDDEN_NODES];
    float biasH[HIDDEN_NODES];
    float weightsHO[HIDDEN_NODES][OUTPUT_NODES];
    float biasO[OUTPUT_NODES];
};

EdgeAI::EdgeAI() 
    : datasetSize(0),
      datasetIndex(0),
      autoMode(true),
      lastLoss(0.0f) {}

void EdgeAI::begin() {
    // Load Auto Mode state
    preferences.begin(NVS_NAMESPACE, false); // Open in read-write
    autoMode = preferences.getBool(KEY_AUTO_MODE, true);
    preferences.end();
    
    // Load Weights from NVS (or generate randomly if not found)
    loadWeightsFromNVS();

    Serial.println("[EdgeAI] Initialized.");
    Serial.print("[EdgeAI] Auto Mode: ");
    Serial.println(autoMode ? "ENABLED" : "DISABLED");
}

float EdgeAI::sigmoid(float x) const {
    return 1.0f / (1.0f + exp(-x));
}

float EdgeAI::sigmoidDerivative(float x) const {
    return x * (1.0f - x);
}

void EdgeAI::initWeightsRandomly() {
    randomSeed(esp_random());

    for (int i = 0; i < INPUT_NODES; i++) {
        for (int j = 0; j < HIDDEN_NODES; j++) {
            weightsIH[i][j] = ((float)random(0, 2000) / 1000.0f) - 1.0f; // -1.0 to 1.0
        }
    }
    for (int j = 0; j < HIDDEN_NODES; j++) {
        biasH[j] = ((float)random(0, 2000) / 1000.0f) - 1.0f;
        for (int k = 0; k < OUTPUT_NODES; k++) {
            weightsHO[j][k] = ((float)random(0, 2000) / 1000.0f) - 1.0f;
        }
    }
    biasO[0] = ((float)random(0, 2000) / 1000.0f) - 1.0f;
    
    lastLoss = 99.9f;
    Serial.println("[EdgeAI] Model weights initialized randomly.");
}

float EdgeAI::predict(float moisturePercent, float tempC, float humidityPercent) {
    float inputs[INPUT_NODES];
    inputs[0] = moisturePercent / 100.0f;   // 0.0 to 1.0
    inputs[1] = tempC / 50.0f;              // Normal range: 0.0 to 1.0 (max 50C)
    inputs[2] = humidityPercent / 100.0f;   // 0.0 to 1.0

    float hidden[HIDDEN_NODES];
    for (int j = 0; j < HIDDEN_NODES; j++) {
        float sum = biasH[j];
        for (int i = 0; i < INPUT_NODES; i++) {
            sum += inputs[i] * weightsIH[i][j];
        }
        hidden[j] = sigmoid(sum);
    }

    float sumO = biasO[0];
    for (int j = 0; j < HIDDEN_NODES; j++) {
        sumO += hidden[j] * weightsHO[j][0];
    }
    float prediction = sigmoid(sumO);

    return prediction;
}

void EdgeAI::4addSample(float moisturePercent, float tempC, float humidityPercent, float label) {
    dataset[datasetIndex].inputs[0] = moisturePercent / 100.0f;
    dataset[datasetIndex].inputs[1] = tempC / 50.0f;
    dataset[datasetIndex].inputs[2] = humidityPercent / 100.0f;
    dataset[datasetIndex].label = constrain(label, 0.0f, 1.0f);

    Serial.print("[EdgeAI] Added sample at index ");
    Serial.print(datasetIndex);
    Serial.print(": [Moisture: ");
    Serial.print(moisturePercent);
    Serial.print("%, Temp: ");
    Serial.print(tempC);
    Serial.print("C, Humidity: ");
    Serial.print(humidityPercent);
    Serial.print("%] -> Label: ");
    Serial.println(label);

    datasetIndex = (datasetIndex + 1) % MAX_SAMPLES;
    if (datasetSize < MAX_SAMPLES) {
        datasetSize++;
    }
}

float EdgeAI::train(int epochs, float learningRate) {
    if (datasetSize == 0) {
        Serial.println("[EdgeAI] Training aborted: No dataset samples available.");
        return 99.9f;
    }

    Serial.print("[EdgeAI] Starting training. Dataset size: ");
    Serial.print(datasetSize);
    Serial.print(". Epochs: ");
    Serial.println(epochs);

    unsigned long startTime = millis();
    float totalMse = 0.0f;

    for (int epoch = 0; epoch < epochs; epoch++) {
        totalMse = 0.0f;
        
        for (int s = 0; s < datasetSize; s++) {
            // Forward pass
            float inputs[INPUT_NODES];
            for (int i = 0; i < INPUT_NODES; i++) {
                inputs[i] = dataset[s].inputs[i];
            }
            float target = dataset[s].label;

            float hidden[HIDDEN_NODES];
            for (int j = 0; j < HIDDEN_NODES; j++) {
                float sum = biasH[j];
                for (int i = 0; i < INPUT_NODES; i++) {
                    sum += inputs[i] * weightsIH[i][j];
                }
                hidden[j] = sigmoid(sum);
            }

            float sumO = biasO[0];
            for (int j = 0; j < HIDDEN_NODES; j++) {
                sumO += hidden[j] * weightsHO[j][0];
            }
            float output = sigmoid(sumO);

            float error = target - output;
            totalMse += error * error;

            float dOutput = error * sigmoidDerivative(output);

            float dHidden[HIDDEN_NODES];
            for (int j = 0; j < HIDDEN_NODES; j++) {
                dHidden[j] = dOutput * weightsHO[j][0] * sigmoidDerivative(hidden[j]);
            }

            for (int j = 0; j < HIDDEN_NODES; j++) {
                weightsHO[j][0] += learningRate * dOutput * hidden[j];
            }
            biasO[0] += learningRate * dOutput;

            for (int i = 0; i < INPUT_NODES; i++) {
                for (int j = 0; j < HIDDEN_NODES; j++) {
                    weightsIH[i][j] += learningRate * dHidden[j] * inputs[i];
                }
            }
            for (int j = 0; j < HIDDEN_NODES; j++) {
                biasH[j] += learningRate * dHidden[j];
            }
        }
        
        totalMse /= datasetSize;
    }

    lastLoss = totalMse;
    unsigned long duration = millis() - startTime;
    
    Serial.print("[EdgeAI] Training complete in ");
    Serial.print(duration);
    Serial.print(" ms. Final Mean Squared Error Loss: ");
    Serial.println(lastLoss, 6);

    saveWeightsToNVS();
    
    return lastLoss;
}

void EdgeAI::resetModel() {
    initWeightsRandomly();
    saveWeightsToNVS();
    datasetSize = 0;
    datasetIndex = 0;
    Serial.println("[EdgeAI] Model reset complete (Weights re-randomized & buffer cleared).");
}

void EdgeAI::setAutoMode(bool enable) {
    autoMode = enable;
    preferences.begin(NVS_NAMESPACE, false);
    preferences.putBool(KEY_AUTO_MODE, autoMode);
    preferences.end();
    Serial.print("[EdgeAI] Auto Mode set to: ");
    Serial.println(autoMode ? "ENABLED" : "DISABLED");
}

void EdgeAI::saveWeightsToNVS() {
    ModelWeights w;
    memcpy(w.weightsIH, weightsIH, sizeof(weightsIH));
    memcpy(w.biasH, biasH, sizeof(biasH));
    memcpy(w.weightsHO, weightsHO, sizeof(weightsHO));
    memcpy(w.biasO, biasO, sizeof(biasO));

    preferences.begin(NVS_NAMESPACE, false);
    preferences.putBytes(KEY_AI_WEIGHTS, &w, sizeof(ModelWeights));
    preferences.end();
    
    Serial.println("[EdgeAI] Model weights successfully saved to NVS.");
}

void EdgeAI::loadWeightsFromNVS() {
    ModelWeights w;
    preferences.begin(NVS_NAMESPACE, true); // Read-only
    
    size_t length = preferences.getBytesLength(KEY_AI_WEIGHTS);
    if (length == sizeof(ModelWeights)) {
        preferences.getBytes(KEY_AI_WEIGHTS, &w, sizeof(ModelWeights));
        memcpy(weightsIH, w.weightsIH, sizeof(weightsIH));
        memcpy(biasH, w.biasH, sizeof(biasH));
        memcpy(weightsHO, w.weightsHO, sizeof(weightsHO));
        memcpy(biasO, w.biasO, sizeof(biasO));
        
        lastLoss = 0.01f; 
        Serial.println("[EdgeAI] Model weights successfully loaded from NVS.");
    } else {
        Serial.println("[EdgeAI] No saved weights found in NVS. Generating initial random weights.");
        preferences.end();
        initWeightsRandomly();
        saveWeightsToNVS();
        return;
    }
    preferences.end();
}

String EdgeAI::getWeightsString() const {
    String out = "";
    for (int i = 0; i < INPUT_NODES; i++) {
        for (int j = 0; j < HIDDEN_NODES; j++) {
            out += String(weightsIH[i][j], 2) + ",";
        }
    }
    for (int j = 0; j < HIDDEN_NODES; j++) {
        out += String(weightsHO[j][0], 2) + ",";
    }
    out += String(biasO[0], 2);
    return out;
}
