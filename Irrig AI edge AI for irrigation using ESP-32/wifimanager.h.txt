#ifndef WIFI_MANAGER_H
#define WIFI_MANAGER_H

#include <Arduino.h>
#include <WiFi.h>
#include <HTTPClient.h>
#include <Preferences.h>
#include "config.h"

class WiFiManager {
public:
    WiFiManager();

    // Load credentials from NVS and initialize WiFi
    void begin();

    // Check WiFi status and handle non-blocking reconnect attempts
    void update(bool bleConnected);

    // Refresh credentials dynamically from NVS (e.g. after a BLE write)
    void loadCredentials();

    // Upload sensor data and pump states to ThingSpeak Cloud
    bool uploadData(float temp, float humid, float moisturePct, bool pumpRunning, float aiLoss, int sampleCount);

    // Getters for connection details
    bool isConnected() const { return connected; }
    String getSSID() const { return ssid; }

private:
    Preferences preferences;
    
    String ssid;
    String password;
    String thingspeakWriteKey;

    bool connected;
    unsigned long lastReconnectAttempt;
    unsigned long reconnectInterval;

    void startConnection();
};

#endif // WIFI_MANAGER_H
